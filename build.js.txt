const esc = (s) => s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
const BODY_SIZE = 13.5;
const ITEM_SIZE = 11.5;
const MONO_W = 0.6;
const sansW = (s, size) => s.length * size * 0.52;

function luminance(hex) {
  const [r, g, b] = [0, 2, 4].map((i) => {
    const c = parseInt(hex.slice(i, i + 2), 16) / 255;
    return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
  });
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
}

function iconColor(name, icon, t) {
  if (t.icon[name]) return t.icon[name];
  const L = luminance(icon.hex);
  if (t.name === 'light' && L > 0.7) return t.muted;
  if (t.name === 'dark' && L < 0.06) return t.body;
  return '#' + icon.hex;
}

function itemWidth(name) {
  return (TECH[name] ? 17 : 0) + name.length * ITEM_SIZE * MONO_W;
}

// Flows a category's items onto rows, separated by quiet middots; brand marks only for real tools.
function items(list, x, y, width, t) {
  const SEP = 15;
  let cx = x, row = 0, svg = '';
  list.forEach((name, i) => {
    const w = itemWidth(name);
    if (i && cx + SEP + w > x + width) { cx = x; row++; }
    else if (i) {
      svg += `<circle cx="${(cx + SEP / 2).toFixed(1)}" cy="${y + row * 20 - 4}" r="1.1" fill="${t.faint}"/>`;
      cx += SEP;
    }
    const by = y + row * 20;
    const icon = TECH[name];
    let tx = cx;
    if (icon) {
      svg += `<path transform="translate(${cx.toFixed(1)},${by - 10.5}) scale(.5)" fill="${iconColor(name, icon, t)}" d="${icon.path}"/>`;
      tx += 17;
    }
    const isTool = name in TECH;
    svg += `<text x="${tx.toFixed(1)}" y="${by}" font-family="${MONO}" font-size="${ITEM_SIZE}" fill="${isTool ? t.body : t.muted}">${esc(name)}</text>`;
    cx += w;
  });
  return { svg, h: row * 20 + 14 };
}

function group(g, x, y, width, t) {
  let svg = `<text x="${x}" y="${y}" font-family="${SANS}" font-size="12" font-weight="600" fill="${t.ink}">${esc(g.label)}</text>`;
  const row = items(g.items, x, y + 20, width, t);
  return { svg: svg + row.svg, h: 20 + row.h };
}

function groups(list, x, y, width, t) {
  let svg = '', cy = y;
  list.forEach((g, i) => {
    if (i) cy += 14;
    const out = group(g, x, cy, width, t);
    svg += out.svg;
    cy += out.h;
  });
  return { svg, h: cy - y };
}

function titleLine(title, aside, x, y, width, size, weight, t) {
  const open = `<text x="${x}" y="${y}" font-family="${SANS}" font-size="${size}" font-weight="${weight}" fill="${t.ink}" letter-spacing="-.1">`;
  if (!aside) return { svg: `${open}${esc(title)}</text>`, h: 0 };
  if (sansW(title, size) + sansW(' · ' + aside, size - 1) <= width) {
    return { svg: `${open}${esc(title)}<tspan font-weight="400" font-size="${size - 1}" fill="${t.muted}"> · ${esc(aside)}</tspan></text>`, h: 0 };
  }
  return {
    svg: `${open}${esc(title)}</text><text x="${x}" y="${y + 20}" font-family="${SANS}" font-size="${size - 2}" fill="${t.muted}">${esc(aside)}</text>`,
    h: 20,
  };
}

  const text = m.pre + m.strong + m.post;

  svg += `<rect x="${px}" y="${y - 7}" width="${w.toFixed(1)}" height="20" rx="10" fill="${accent}" fill-opacity=".08" stroke="${accent}" stroke-opacity=".35"/>`;

function role(r, x, y, width, accent, t) {
  let svg = '';
  const DATE_SIZE = 11.5, LS = 0.2;
  const dateW = r.date.length * (DATE_SIZE * MONO_W + LS);
  svg += `<text x="${x}" y="${y + 10}" font-family="${MONO}" font-size="${DATE_SIZE}" letter-spacing="${LS}" fill="${accent}" font-weight="600">${esc(r.date)}</text>`;
  let dateEnd = x + dateW;
  if (r.note) {
    const nx = x + dateW + 7;
    const nw = r.note.length * 6 + 14;
    svg += `<rect x="${nx.toFixed(1)}" y="${y - 1}" width="${nw}" height="15" rx="7.5" fill="none" stroke="${t.line}"/>`;
    svg += `<text x="${(nx + nw / 2).toFixed(1)}" y="${y + 9.5}" text-anchor="middle" font-family="${MONO}" font-size="10" fill="${t.muted}">${esc(r.note)}</text>`;
    dateEnd = nx + nw;
  }
  if (r.live) {
    const lx = (x + dateW + 9).toFixed(1);
    svg += `<circle cx="${lx}" cy="${y + 6}" r="3" fill="${accent}"/>`;
    svg += `<circle class="motion" cx="${lx}" cy="${y + 6}" r="3" fill="none" stroke="${accent}"><animate attributeName="r" values="3;8;8" keyTimes="0;.6;1" dur="2.4s" repeatCount="indefinite"/><animate attributeName="opacity" values=".7;0;0" keyTimes="0;.6;1" dur="2.4s" repeatCount="indefinite"/></circle>`;
    dateEnd += 14;
  }
  const tl = titleLine(r.title, r.aside, x, y + 33, width, 15.5, 600, t);
  svg += tl.svg;
  let cy = y + 33 + tl.h;
  if (r.context) {
    cy += 21;
    svg += `<text x="${x}" y="${cy}" font-family="${SANS}" font-size="${BODY_SIZE}" fill="${t.body}">${esc(r.context)}</text>`;
  }
  if (r.metric) {
    cy += 22;
    svg += metric(r.metric, x, cy - 4, accent, t);
    cy += 6;
  }
  cy += 26;
  const gs = groups(r.groups, x, cy, width, t);
  svg += gs.svg;
  cy += gs.h;
  return { svg, h: cy - y, nodeY: y + 6, dateStart: x, dateEnd };
}

function section(sec, x, y, width, t) {
  const accent = t[sec.key];
  let svg = header(sec.key, sec.title, sec.span, x, y, width, accent, t);
  const nodes = [{ y: y - 4, kind: 'section', accent }];
  let cy = y + 42;
  sec.orgs.forEach((org, oi) => {
    if (oi) cy += 22;
    const ol = titleLine(org.name, org.alt, x, cy + 14, width, 18, 600, t);
    svg += ol.svg;
    cy += ol.h;
    if (org.sub) {
      svg += `<text x="${x}" y="${cy + 34}" font-family="${SANS}" font-size="13" fill="${t.muted}">${esc(org.sub)}</text>`;
      cy += 58;
    } else cy += 38;
    org.roles.forEach((r, ri) => {
      if (ri) cy += 30;
      const out = role(r, x, cy, width, accent, t);
      svg += out.svg;
      nodes.push({ y: out.nodeY, kind: 'role', accent, dateStart: out.dateStart, dateEnd: out.dateEnd });
      cy += out.h;
    });
  });
  return { svg, endY: cy, nodes };
}

function shared(x, y, width, cols, t) {
  let svg = header('tools', SHARED.title, '', x, y, width, t.muted, t);
  const gap = 28;
  const colW = (width - gap * (cols - 1)) / cols;
  let cy = y + 42, rowH = 0;
  SHARED.groups.forEach((g, i) => {
    const col = i % cols;
    if (i && col === 0) { cy += rowH + 16; rowH = 0; }
    const out = group(g, x + col * (colW + gap), cy, colW, t);
    svg += out.svg;
    rowH = Math.max(rowH, out.h);
  });
  return { svg, endY: cy + rowH };
}

// Curve sampled as a fine polyline so arc length (used to sync node glows with the pulse) is exact.
function buildCurve(cx, y0, y1, amp, period) {
  const pts = [];
  for (let y = y0; y <= y1; y += 3) {
    const ease = Math.min(1, (y - y0) / 40);
    const fade = Math.min(1, (y1 - y) / 40);
    pts.push([cx + amp * Math.sin(((y - y0) / period) * Math.PI * 2) * ease * fade, y]);
  }
  let d = `M${pts[0][0].toFixed(2)} ${pts[0][1]}`;
  const lens = [0];
  for (let i = 1; i < pts.length; i++) {
    d += `L${pts[i][0].toFixed(2)} ${pts[i][1]}`;
    lens.push(lens[i - 1] + Math.hypot(pts[i][0] - pts[i - 1][0], pts[i][1] - pts[i - 1][1]));
  }
  const total = lens[lens.length - 1];
  const at = (y) => {
    const i = Math.max(0, Math.min(pts.length - 1, Math.round((y - y0) / 3)));
    return { x: pts[i][0], f: lens[i] / total };
  };
  return { d, total, at };
}

function stream(curve, t, dur) {
  let svg = `<g class="motion">`;
  svg += `<path d="${curve.d}" fill="none" stroke="${t.exp}" stroke-opacity=".9" stroke-width="2" stroke-linecap="round" stroke-dasharray="48 ${Math.ceil(curve.total + 120)}" stroke-dashoffset="48"><animate attributeName="stroke-dashoffset" from="48" to="${-Math.ceil(curve.total)}" dur="${dur}s" repeatCount="indefinite"/></path>`;
  const bits = '1011001110100101';
  const n = bits.length;
  for (let i = 0; i < n; i++) {
    const begin = -((dur / n) * i).toFixed(2);
    const side = i % 2 ? 5 : -11;
    const col = i % 3 === 0 ? t.edu : t.exp;
    svg += `<text x="${side}" y="3" font-family="${MONO}" font-size="9" fill="${col}" opacity="0">${bits[i]}<animateMotion dur="${dur}s" begin="${begin}s" repeatCount="indefinite"><mpath href="#railPath"/></animateMotion><animate attributeName="opacity" values="0;.8;.55;0" keyTimes="0;.06;.75;1" dur="${dur}s" begin="${begin}s" repeatCount="indefinite"/></text>`;
  }
  return svg + `</g>`;
}

const fx = (v) => v.toFixed(3);

// A synapse branch: rail node -> gentle arc -> terminal bouton, a small cleft, then the date.
// Packets leave the node exactly when the rail pulse reaches it, carry 0/1 bits along the branch,
// and dissolve at the terminal while the date briefly lights up, as if the bits became the data.
function synapse(n, x, f, t, dur, id) {
  const right = n.dateStart > x;
  const x1 = right ? n.dateStart - 9 : n.dateEnd + 9;
  const len = Math.abs(x1 - x);
  const mid = (x + x1) / 2;
  const d = `M${fx(x)} ${n.y}Q${fx(mid)} ${n.y - Math.min(7, len * 0.06)} ${fx(x1)} ${n.y}`;
  const tb = Math.min(0.995 - f - 0.1, Math.max(0.5 / dur, len / 300 / dur));
  const a = f, b = f + tb;
  let svg = `<path id="${id}" d="${d}" fill="none" stroke="${t.line}" stroke-width="1"/>`;
  svg += `<circle cx="${fx(x1)}" cy="${n.y}" r="2" fill="${n.accent}" fill-opacity=".55"/>`;

  const hx = (right ? n.dateStart : n.dateStart) - 5;
  const hw = n.dateEnd - n.dateStart + 10;
  svg += `<g class="motion">`;
  svg += `<rect x="${fx(hx)}" y="${n.y - 10}" width="${fx(hw)}" height="20" rx="5" fill="${n.accent}" opacity="0"><animate attributeName="opacity" values="0;0;.16;0;0" keyTimes="0;${fx(b - 0.005)};${fx(b + 0.02)};${fx(Math.min(0.999, b + 0.12))};1" dur="${dur}s" repeatCount="indefinite"/></rect>`;
  svg += `<path d="${d}" fill="none" stroke="${n.accent}" stroke-width="1.4" stroke-linecap="round" stroke-dasharray="14 ${Math.ceil(len + 40)}" stroke-dashoffset="14" opacity="0"><animate attributeName="stroke-dashoffset" values="14;14;${-Math.ceil(len)};${-Math.ceil(len)}" keyTimes="0;${fx(a)};${fx(b)};1" dur="${dur}s" repeatCount="indefinite"/><animate attributeName="opacity" values="0;0;.9;.9;0;0" keyTimes="0;${fx(a)};${fx(a + 0.005)};${fx(b - 0.005)};${fx(b)};1" dur="${dur}s" repeatCount="indefinite"/></path>`;
  const bits = ['1', '0', '1', '1'];
  bits.forEach((bit, k) => {
    const s = a + k * tb * 0.12;
    const e = Math.min(0.998, s + tb);
    const dy = k % 2 ? 9 : -4;
    svg += `<text x="-3" y="${dy}" font-family="${MONO}" font-size="8.5" fill="${n.accent}" opacity="0">${bit}<animateMotion dur="${dur}s" repeatCount="indefinite" calcMode="linear" keyPoints="0;0;1;1" keyTimes="0;${fx(s)};${fx(e)};1"><mpath href="#${id}"/></animateMotion><animate attributeName="opacity" values="0;0;.85;.85;0;0" keyTimes="0;${fx(s)};${fx(s + 0.01)};${fx(e - tb * 0.35)};${fx(e)};1" dur="${dur}s" repeatCount="indefinite"/></text>`;
  });
  svg += `<circle cx="${fx(x1)}" cy="${n.y}" r="2" fill="none" stroke="${n.accent}" opacity="0"><animate attributeName="r" values="2;2;7;7" keyTimes="0;${fx(b)};${fx(Math.min(0.999, b + 0.08))};1" dur="${dur}s" repeatCount="indefinite"/><animate attributeName="opacity" values="0;0;.6;0;0" keyTimes="0;${fx(b - 0.002)};${fx(b)};${fx(Math.min(0.999, b + 0.08))};1" dur="${dur}s" repeatCount="indefinite"/></circle>`;
  return svg + `</g>`;
}

function nodeMarks(curve, nodes, t, dur) {
  let svg = '';
  nodes.forEach((n, i) => {
    if (n.kind === 'section') return;
    const { x, f } = curve.at(n.y);
    const xs = x.toFixed(2);
    svg += synapse(n, x, f, t, dur, `syn${i}`);
    const k = [0, Math.max(0.001, f - 0.002), Math.min(0.998, f + 0.03), Math.min(0.999, f + 0.16), 1].map((v) => v.toFixed(3)).join(';');
    svg += `<circle class="motion" cx="${xs}" cy="${n.y}" r="4" fill="${n.accent}" opacity="0"><animate attributeName="opacity" values="0;0;.3;0;0" keyTimes="${k}" dur="${dur}s" repeatCount="indefinite"/><animate attributeName="r" values="4;4;10;12;4" keyTimes="${k}" dur="${dur}s" repeatCount="indefinite"/></circle>`;
    svg += `<circle cx="${xs}" cy="${n.y}" r="4" fill="${t.node}" stroke="${n.accent}" stroke-width="1.6"/>`;
    svg += `<circle cx="${xs}" cy="${n.y}" r="1.4" fill="${n.accent}"/>`;
  });
  return svg;
}

function defs(t, y0, y1) {
  return `<defs><linearGradient id="railGrad" x1="0" y1="${y0}" x2="0" y2="${y1}" gradientUnits="userSpaceOnUse"><stop offset="0" stop-color="${t.exp}" stop-opacity=".75"/><stop offset=".22" stop-color="${t.line}"/><stop offset=".9" stop-color="${t.line}"/><stop offset="1" stop-color="${t.line}" stop-opacity="0"/></linearGradient></defs>
<style>@media (prefers-reduced-motion: reduce){.motion{display:none}}</style>`;
}
